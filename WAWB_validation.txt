! Purpose
!   Validates the cache controller simulator for the
!   Write-Allocate / Write-Back (WAWB) policy.
!
! Command line used for validation:
!   cachesim WAWB_validation.txt WAWB 256 16
!
!   write_policy    = WAWB
!   blocks_in_cache = 256          (8 index bits)
!   words_in_block  = 16           (4 offset bits)
!   tag bits        = 32 - 8 - 4 = 20
!
! Address layout (32 bits):
!   bits 31 .. 12 : Tag             (5 hex digits)
!   bits 11 ..  4 : Cache block ID  (2 hex digits)
!   bits  3 ..  0 : Block offset    (1 hex digit)
!
! Decision paths exercised (all 8 WAWB branches):
!   1. Read  hit                                          (RH)
!   2. Read  miss on invalid block                        (RM-invalid)
!   3. Read  miss evicting valid+clean block              (RM-valid-clean)
!   4. Read  miss evicting valid+dirty block (write back) (RM-valid-dirty)
!   5. Write hit (sets dirty bit)                         (WH)
!   6. Write miss on invalid block                        (WM-invalid)
!   7. Write miss evicting valid+clean block              (WM-valid-clean)
!   8. Write miss evicting valid+dirty block (write back) (WM-valid-dirty)
!
! Expected final simulator output line:
!   8 6 112 32 5 3 2 4 16 256 WAWB_validation.txt WAWB
!     CPUR=8  CPUW=6  NRA=112  NWA=32
!     NCRH=5  NCRM=3  NCWH=2   NCWM=4   WIB=16  BIC=256
!
!   All ten numeric output values are distinct, so an accidental swap of any
!   two counters in the simulator would change the output line.
! ============================================================================


! ----------------------------------------------------------------------------
! Section 1: cold read miss (RM-invalid)
! Cache block 0x00 is invalid -> read miss, load block, no write back.
!   counters: NCRM=1, NRA=16
! ----------------------------------------------------------------------------
R 00001000 0000


! ----------------------------------------------------------------------------
! Section 2: read hits in the same block (RH)
! Block 0x00 now holds tag 0x00001 and is clean.
!   counters: NCRH=1, NCRH=2  (cumulative 1, 2)
! ----------------------------------------------------------------------------
R 00001001 0000
R 00001002 0000


! ----------------------------------------------------------------------------
! Section 3: read miss evicting a valid+clean block (RM-valid-clean)
! Block 0x00 holds tag 0x00001, dirty=0.  New address has tag 0x00002.
! No write back; load block with tag 0x00002.
!   counters: NCRM=2, NRA=32
! ----------------------------------------------------------------------------
R 00002000 0000


! ----------------------------------------------------------------------------
! Section 4: more read hits (RH)
!   counters: NCRH=3, NCRH=4
! ----------------------------------------------------------------------------
R 00002001 0000
R 00002002 0000


! ----------------------------------------------------------------------------
! Section 5: write hit (WH)
! Block 0x00 has tag 0x00002 and is clean.  Writing to it sets the dirty bit.
! No external memory access (write back policy).
!   counters: NCWH=1
! ----------------------------------------------------------------------------
W 00002003 ABCD


! ----------------------------------------------------------------------------
! Section 6: read miss evicting valid+dirty block (RM-valid-dirty)
! Block 0x00 has tag 0x00002, dirty=1 -> the 16 words must be written back to
! external memory before the new block (tag 0x00003) is loaded.
!   counters: NCRM=3, NWA=16, NRA=48
! ----------------------------------------------------------------------------
R 00003000 0000


! ----------------------------------------------------------------------------
! Section 7: read hit on the freshly loaded clean block (RH)
!   counters: NCRH=5
! ----------------------------------------------------------------------------
R 00003001 0000


! ----------------------------------------------------------------------------
! Section 8: write miss evicting valid+clean block (WM-valid-clean)
! Block 0x00 holds tag 0x00003, dirty=0.  Write to a different tag (0x00004)
! triggers a write miss with no write back.  Block becomes dirty.
!   counters: NCWM=1, NRA=64
! ----------------------------------------------------------------------------
W 00004000 1111


! ----------------------------------------------------------------------------
! Section 9: write hit on the new dirty block (WH)
!   counters: NCWH=2
! ----------------------------------------------------------------------------
W 00004001 2222


! ----------------------------------------------------------------------------
! Section 10: write miss evicting valid+dirty block (WM-valid-dirty)
! Block 0x00 holds tag 0x00004, dirty=1.  A write to tag 0x00005 forces a
! write back of the 16 words and load of the new block.
!   counters: NCWM=2, NWA=32, NRA=80
! ----------------------------------------------------------------------------
W 00005000 3333


! ----------------------------------------------------------------------------
! Section 11: write miss on a different invalid block (WM-invalid)
! Block 0x01 has never been touched -> cold write miss.  No write back.
!   counters: NCWM=3, NRA=96
! ----------------------------------------------------------------------------
W 00000010 4444


! ----------------------------------------------------------------------------
! Section 12: another write miss on a different invalid block (WM-invalid)
! Block 0x02 cold miss.  Adds the 4th write miss so that the final NCWM value
! (=4) is distinct from every other counter.
!   counters: NCWM=4, NRA=112
! ----------------------------------------------------------------------------
W 00000020 5555
